MainWindow::MainWindow(QWidget *parent)
    : QMainWindow(parent)
    , ui(new Ui::MainWindow)
{
    ui->setupUi(this);
    createMenus();
    setupUI();
    setCentralWidget(centralWidget);
    setWindowTitle("Qt Chess");
    resize(800, 800);
}

MainWindow::~MainWindow()
{
    delete ui;
}

void MainWindow::setupUI() {
    centralWidget = new QWidget(this);
    QVBoxLayout* mainLayout = new QVBoxLayout(centralWidget);
    
    QGraphicsView* view = new QGraphicsView;
    mainLayout->addWidget(view);
    
    setupBottomPanel();
}

void MainWindow::setupBottomPanel() {
    QWidget* bottomPanel = new QWidget;
    bottomPanel->setFixedHeight(80);
    QHBoxLayout* bottomLayout = new QHBoxLayout(bottomPanel);
    bottomLayout->setContentsMargins(5, 5, 5, 5);
    bottomLayout->setSpacing(10);

    moveHistory = new QTextEdit;
    moveHistory->setReadOnly(true);
    moveHistory->setMaximumHeight(60);
    bottomLayout->addWidget(moveHistory);

    gameStatus = new QLabel("White to move");
    bottomLayout->addWidget(gameStatus);

    difficultySelect = new QComboBox;
    difficultySelect->addItems({"Easy", "Medium", "Hard"});
    bottomLayout->addWidget(difficultySelect);

    confirmMove = new QPushButton("Confirm Move");
    bottomLayout->addWidget(confirmMove);

    connect(confirmMove, &QPushButton::clicked, this, &MainWindow::handleConfirmMove);
    connect(difficultySelect, QOverload<int>::of(&QComboBox::currentIndexChanged),
            this, &MainWindow::setDifficulty);

    centralWidget->layout()->addWidget(bottomPanel);
}

void MainWindow::createMenus() {
    QMenu* gameMenu = menuBar()->addMenu(tr("&Game"));
    QAction* newGameAct = new QAction(tr("&New Game"), this);
    connect(newGameAct, &QAction::triggered, this, &MainWindow::newGame);
    gameMenu->addAction(newGameAct);
}
